# Sample Code:
```
python
import openai
from google import genai
import os

# Set up API keys (Replace with your actual keys)
OPENAI_API_KEY = "your-openai-api-key"
GEMINI_API_KEY = "your-gemini-api-key"

# Initialize ChatGPT
openai.api_key = OPENAI_API_KEY

# Initialize Gemini
os.environ["API_KEY"] = GEMINI_API_KEY
gemini_client = genai.Client(api_key=os.environ["API_KEY"])

# Function to get answer from ChatGPT
def get_chatgpt_answer(question):
    try:
        response = openai.ChatCompletion.create(
            model="gpt-3.5-turbo",
            messages=[
                {"role": "system", "content": "You are a helpful AI assistant."},
                {"role": "user", "content": question}
            ],
            max_tokens=150,
            temperature=0.7
        )
        return response.choices.message['content'].strip()
    except Exception as e:
        return f"Error: {str(e)}"

# Function to get answer from Gemini
def get_gemini_answer(question):
    try:
        response = gemini_client.models.generate_content(
            model='gemini-2.0-flash',
            contents=question
        )
        return response.text
    except Exception as e:
        return f"Error: {str(e)}"

# Function to compare answers from both models
def compare_answers(question):
    print(f"\n{'='*70}")
    print(f"Question: {question}")
    print(f"{'='*70}\n")
    
    # Get answers from both models
    answer_chatgpt = get_chatgpt_answer(question)
    answer_gemini = get_gemini_answer(question)
    
    # Display results
    print(f"ChatGPT Answer:\n{'-'*70}")
    print(answer_chatgpt)
    print(f"\n{'='*70}\n")
    
    print(f"Gemini Answer:\n{'-'*70}")
    print(answer_gemini)
    print(f"\n{'='*70}\n")
    
    # Compare answers
    if answer_chatgpt.lower() == answer_gemini.lower():
        summary = "Both models provided identical answers."
    else:
        summary = "The answers are different. Each model has its unique response style."
    
    print(f"Summary: {summary}\n")
    
    return {
        "question": question,
        "chatgpt_answer": answer_chatgpt,
        "gemini_answer": answer_gemini,
        "summary": summary
    }

# Run the comparison with sample questions
if __name__ == "__main__":
    # Test Question 1
    question1 = "What are the benefits of renewable energy?"
    result1 = compare_answers(question1)
    
    # Test Question 2 (Optional)
    question2 = "Explain quantum computing in simple terms."
    result2 = compare_answers(question2)
    
    # Test Question 3 (Optional)
    question3 = "Write a Python function to check if a number is prime."
    result3 = compare_answers(question3)
    
    print("\n" + "="*70)
    print("EXPERIMENT COMPLETED SUCCESSFULLY")
    print("="*70)
```
